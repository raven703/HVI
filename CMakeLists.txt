if(MINGW)
 # Make the demo launchable from Explorer or a clean PowerShell, without a
 # toolchain PATH. Keep a single shared C++/thread runtime for DLL COM objects.
 foreach(runtime_dll libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=${runtime_dll}
   OUTPUT_VARIABLE runtime_path OUTPUT_STRIP_TRAILING_WHITESPACE)
  # Some WinLibs distributions keep DLLs beside the compiler but GCC's
  # library search returns only the unchanged filename for these DLLs.
  if(NOT IS_ABSOLUTE "${runtime_path}" OR NOT EXISTS "${runtime_path}")
   get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
   set(runtime_path "${compiler_bin}/${runtime_dll}")
  endif()
  if(EXISTS "${runtime_path}")
   add_custom_command(TARGET hvi_runtime POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${runtime_path}" "$<TARGET_FILE_DIR:hvi_runtime>")
  else()
   message(FATAL_ERROR "Required MinGW runtime not found: ${runtime_dll}")
  endif()
 endforeach()
endif()
