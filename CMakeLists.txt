cmake_minimum_required(VERSION 3.20)
project(HunterVirtualInput LANGUAGES CXX)
if(NOT WIN32)
 message(FATAL_ERROR "Windows is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hvi INTERFACE)
target_include_directories(hvi INTERFACE include)
target_compile_definitions(hvi INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX)
foreach(target controller receiver_a receiver_b unit_tests)
 if(target MATCHES "receiver")
  add_executable(${target} src/receiver.cpp)
 else()
  add_executable(${target} src/${target}.cpp)
 endif()
 target_link_libraries(${target} PRIVATE hvi)
endforeach()
enable_testing()
add_test(NAME state_and_routing COMMAND unit_tests)
add_executable(api_probe src/api_probe.cpp)
target_link_libraries(api_probe PRIVATE hvi dinput8 dxguid user32)
add_executable(model_tests src/model_tests.cpp)
target_link_libraries(model_tests PRIVATE hvi)
add_test(NAME keyboard_mouse_buffer_model COMMAND model_tests)
add_executable(test_controller src/test_controller.cpp)
target_link_libraries(test_controller PRIVATE hvi)
add_executable(mouse_tests src/mouse_tests.cpp)
target_link_libraries(mouse_tests PRIVATE hvi dinput8 dxguid)
add_test(NAME directinput_mouse_unit COMMAND mouse_tests)
set_tests_properties(state_and_routing keyboard_mouse_buffer_model directinput_mouse_unit PROPERTIES LABELS "unit")
